"use client";

import { motion } from "framer-motion";
import { useEffect } from "react";
import Image from "next/image";
import { EventBus } from "@/game/events";

export type NotificationType = "SUCCESS" | "GAMEOVER" | "LEVEL_UP" | "VICTORY";

export interface GameNotificationProps {
  type: NotificationType;
  title: string;
  message: string;
  points?: number;
  onConfirm?: () => void;
  confirmLabel?: string;
}

const THEMES = {
  SUCCESS: {
    color: "text-[#FF85A1]", // Strawberry Pink
    border: "border-[#FFB3C6]",
    bg: "bg-[#FFF0F3]",
    btn: "bg-[#FF85A1] hover:bg-[#FF4D6D]",
    shadow: "shadow-[0_12px_0_#FFB3C6]",
    sound: "success",
  },
  GAMEOVER: {
    color: "text-[#EF4444]",
    border: "border-[#FCA5A5]",
    bg: "bg-[#FEE2E2]",
    btn: "bg-[#EF4444] hover:bg-[#DC2626]",
    shadow: "shadow-[0_12px_0_#FCA5A5]",
    sound: "gameover",
  },

  LEVEL_UP: {
    color: "text-[#4CC9F0]", // Sky Blue
    border: "border-[#BEE9E8]",
    bg: "bg-[#F0F9F9]",
    btn: "bg-[#4CC9F0] hover:bg-[#4361EE]",
    shadow: "shadow-[0_12px_0_#BEE9E8]",
    sound: "level-up",
  },
  VICTORY: {
    color: "text-[#FFD60A]", // Honey Yellow
    border: "border-[#FFEA00]",
    bg: "bg-[#FFFDF0]",
    btn: "bg-[#FFD60A] hover:bg-[#FFC300]",
    shadow: "shadow-[0_12px_0_#FFEA00]",
    sound: "victory",
  },
};

export function GameNotification({
  type,
  title,
  message,
  points = 0,
  onConfirm,
  confirmLabel = "OKAY!",
}: GameNotificationProps) {
  const theme = THEMES[type];

  useEffect(() => {
    EventBus.emit("play-ui-sound", theme.sound);
  }, [type, theme.sound]);

  return (
    <div className="absolute inset-0 z-[300] flex items-center justify-center bg-white/30 backdrop-blur-md p-4 font-gluten">
      <motion.div
        initial={{ scale: 0.4, opacity: 0, rotate: -5 }}
        animate={{ scale: 1, opacity: 1, rotate: 0 }}
        exit={{ scale: 0.4, opacity: 0, rotate: 5 }}
        transition={{ type: "spring", stiffness: 200, damping: 15 }}
        className={`relative w-full max-w-sm p-10 rounded-[48px] border-[6px] ${theme.bg} ${theme.border} ${theme.shadow} text-center`}
      >
        <div className="relative z-10 flex flex-col items-center">
          <h2
            className={`text-5xl mb-4 tracking-tight ${theme.color} drop-shadow-sm`}
          >
            {title}
          </h2>

          {/* COIN SECTION: HALF-CIRCLE ANIMATION */}
          {type === "SUCCESS" && (
            <div className="relative w-full h-44 flex items-center justify-center mb-2">
              <div className="absolute w-32 h-32 bg-yellow-300/30 blur-3xl rounded-full" />

              <div className="relative w-full h-full">
                {[...Array(5)].map((_, i) => {
                  // Calculate arc positions (angle from 210 to 330 degrees)
                  const angle = i * 30 + 210;
                  const radius = 85;
                  const x = Math.cos((angle * Math.PI) / 180) * radius;
                  const y = Math.sin((angle * Math.PI) / 180) * radius + 40;

                  return (
                    <motion.div
                      key={i}
                      className="absolute"
                      initial={{ x: 0, y: 50, opacity: 0 }}
                      animate={{
                        x,
                        y: [y, y - 12, y],
                        opacity: 1,
                      }}
                      transition={{
                        delay: i * 0.12,
                        y: { repeat: Infinity, duration: 2, ease: "easeInOut" },
                      }}
                      style={{ left: "calc(50% - 15px)", top: "40%" }}
                    >
                      <Image
                        src="/ui/coin.png"
                        width={30}
                        height={30}
                        style={{ width: "30px", height: "auto" }}
                        alt="coin"
                        priority
                      />
                    </motion.div>
                  );
                })}
              </div>

              <motion.div
                animate={{ scale: [1, 1.08, 1] }}
                transition={{
                  repeat: Infinity,
                  duration: 2,
                  ease: "easeInOut",
                }}
                className="absolute z-10 bottom-4"
              >
                <Image
                  src="/ui/coin-pot.png"
                  width={110}
                  height={90}
                  style={{ width: "110px", height: "auto" }}
                  alt="pot"
                  priority
                />
              </motion.div>
            </div>
          )}

          {/* MESSAGE SECTION */}
          <div className="space-y-4 mb-6">
            <p className="text-[#8B5E66] text-xl font-bold leading-tight px-4">
              {message}
            </p>

            {points > 0 && (
              <motion.div
                initial={{ opacity: 0, scale: 0.8 }}
                animate={{ opacity: 1, scale: 1 }}
                className="block"
              >
                <span className="text-yellow-700 text-lg font-bold bg-yellow-100 px-4 py-1.5 rounded-full border-2 border-yellow-200">
                  You get ${points}
                </span>
              </motion.div>
            )}
          </div>

          {/* SMALLER CONFIRM BUTTON */}
          {onConfirm && (
            <motion.button
              whileHover={{ scale: 1.05, rotate: 1 }}
              whileTap={{ scale: 0.95 }}
              onClick={() => {
                EventBus.emit("play-ui-sound", "click");
                onConfirm();
              }}
              className={`px-8 py-2.5 rounded-full text-white text-xl font-bold transition-all
                ${theme.btn} shadow-[0_5px_0_rgba(0,0,0,0.15)]`}
            >
              {confirmLabel.toUpperCase()}
            </motion.button>
          )}
        </div>

        {/* Decorative Sparkles */}
        <motion.div
          animate={{ opacity: [0.2, 0.8, 0.2] }}
          transition={{ repeat: Infinity, duration: 3 }}
          className="absolute top-8 left-8 text-2xl"
        >
          ✨
        </motion.div>
        <motion.div
          animate={{ opacity: [0.2, 0.8, 0.2] }}
          transition={{ repeat: Infinity, duration: 3, delay: 1 }}
          className="absolute bottom-12 right-8 text-2xl"
        >
          ✨
        </motion.div>
      </motion.div>
    </div>
  );
}
